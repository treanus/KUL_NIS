# dwiprep processing
#  dwiprep_options: space-separated list of FOD methods (dhollander, tax, tournier, lore_sd)

#                      outputs are resampled to 1.3mm afterwards for downstream compatibility
#                  0 = use upsampled data (dwi_preproced.mif, 1.3mm iso)
#  ncpu: numer of cpu's to use for each subject
#  simultaneous: defines how many subjects to process simultaneously
do_dwiprep: 1
dwiprep_options: dhollander lore_sd
#  loresd_env: name of the conda env with lore_sd installed; defaults to $KUL_LORESD_ENV ('lore_sd', the fixed name the KUL_NIS installer creates it under) when left blank -- only set this if you used a different env name
loresd_env:
use_native_dwi: 1
#  synbzero_disco_instead_of_topup / rev_phase_for_topup_only describe the
#  ACQUISITION, not the processing type, so both default to "auto":
#    one phase-encoding direction only  -> synb0=1, revonly=0  (no topup possible)
#    both, reverse phase is b0-only     -> synb0=0, revonly=1  (topup, rev not data)
#    both, reverse phase diffusion-wtd  -> synb0=0, revonly=0  (topup, rev is data)
#  KUL_clinical_fmridti.sh resolves this from the BIDS dwi series after dcm2bids
#  and writes the numbers into the copy in KUL_LOG/. Set 0 or 1 here to override;

#  NOTE: KUL_clinical_fmridti.sh overwrites this with its own -n when it copies
